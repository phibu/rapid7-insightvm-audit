# 0.2.0 Phase 2 — Interactivity Layer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship 0.2.0 — the interactivity layer for the self-contained HTML report. Sticky filter bar with severity chips + search + URL-hash sync, three-state theme toggle (system/light/dark) with `localStorage` persistence and FOUC prevention, `<noscript>` fallback, `prefers-reduced-motion` handling, and a static-HTML a11y test sweep.

**Architecture:** Single Jinja2 template touched (`templates/report.html.j2`) plus one new test file (`tests/test_report_a11y.py`). Native `<details>` retained for rule cards (no JS toggle needed). Filter mechanism is CSS-driven: JS only sets attributes on `<body>`; CSS attribute selectors hide non-matching rule cards. Theme toggle is three-state cycle persisted in `localStorage`. No `report.py`, `config.py`, or HTTP client changes; no new dependencies.

**Tech Stack:** Jinja2 (already in use), pure DOM APIs (no jQuery, no React, no anything), `re` module for the static-HTML a11y test. Python 3.11+, pytest.

### 2d. Body-tail JS for the toggle cycle

- [ ] **Step 2.4: Add the toggle wiring JS just before `</body>`**

Find `</body>` (around line 434, but it shifts with each insertion — use grep). Immediately before it, add:

```html
<script>
(function(){
  var btn = document.querySelector('.theme-toggle');
  if (!btn) return;
  var states = [
    { name: 'system', icon: '#icon-system', label: 'System' },
    { name: 'light',  icon: '#icon-light',  label: 'Light' },
    { name: 'dark',   icon: '#icon-dark',   label: 'Dark' },
  ];
  function apply(state) {
    btn.setAttribute('data-theme-state', state.name);
    btn.setAttribute('aria-label', 'Theme: ' + state.name);
    btn.querySelector('.theme-icon use').setAttribute('href', state.icon);
    btn.querySelector('.theme-label').textContent = state.label;
    if (state.name === 'system') {
      document.documentElement.removeAttribute('data-theme');
      try { localStorage.removeItem('theme'); } catch(e) {}
    } else {
      document.documentElement.setAttribute('data-theme', state.name);
      try { localStorage.setItem('theme', state.name); } catch(e) {}
    }
  }
  // Restore state from <html data-theme> set by the FOUC head script.
  var initial = document.documentElement.getAttribute('data-theme');
  if (initial === 'light') apply(states[1]);
  else if (initial === 'dark') apply(states[2]);
  // (no-op if 'system' — initial markup is already system)
  btn.addEventListener('click', function(){
    var current = btn.getAttribute('data-theme-state');
    var idx = states.findIndex(function(s){ return s.name === current; });
    apply(states[(idx + 1) % states.length]);
  });
})();
</script>
```

- [ ] **Step 4.1: Append severity-chip wiring to the body-tail script**

In the existing `<script>` block from Task 2 (the IIFE that wires the theme toggle, just before `</body>`), append a SECOND IIFE just after the first one (still before `</script>`). Or — more readably — replace the single `<script>` block with two adjacent IIFEs in the same `<script>` tag:

The new code below goes AFTER the closing `})();` of the theme-toggle IIFE, still inside the same `<script>` element:

```javascript
(function(){
  var bar = document.querySelector('.filter-bar');
  if (!bar) return;
  var sevChips = bar.querySelectorAll('.chip[data-severity]');
  function setSeverity(sev) {
    sevChips.forEach(function(c){
      c.setAttribute('aria-pressed', c.getAttribute('data-severity') === sev ? 'true' : 'false');
    });
    if (sev === 'all') {
      document.body.removeAttribute('data-filter-severity');
    } else {
      document.body.setAttribute('data-filter-severity', sev);
    }
    syncHash();
  }
  function syncHash() {
    var parts = [];
    var sev = document.body.getAttribute('data-filter-severity');
    if (sev) parts.push('severity=' + sev);
    if (document.body.hasAttribute('data-filter-changed')) parts.push('changed');
    var search = document.body.getAttribute('data-filter-search-q');
    if (search) parts.push('search=' + encodeURIComponent(search));
    var hash = parts.length ? '#' + parts.join('&') : ' ';
    if (location.hash !== hash) {
      history.replaceState({}, '', hash === ' ' ? location.pathname + location.search : hash);
    }
  }
  sevChips.forEach(function(c){
    c.addEventListener('click', function(){
      setSeverity(c.getAttribute('data-severity'));
    });
  });
  // Restore from hash on load.
  var hash = location.hash.replace(/^#/, '');
  if (hash) {
    var params = {};
    hash.split('&').forEach(function(p){
      var eq = p.indexOf('=');
      if (eq < 0) params[p] = true;
      else params[p.slice(0, eq)] = decodeURIComponent(p.slice(eq + 1));
    });
    if (params.severity && ['all','fail','warn','pass','skipped'].indexOf(params.severity) >= 0) {
      setSeverity(params.severity);
    }
  }
  // Expose syncHash for Tasks 5 & 6 to call after their state changes.
  window.__phase2_syncHash = syncHash;
})();
```

The `window.__phase2_syncHash` global is the bridge to Tasks 5 (search) and 6 (Changed chip). Each task adds its own handler that mutates `<body>` attributes and calls `__phase2_syncHash()` afterward.

```javascript
(function(){
  var input = document.querySelector('.filter-search');
  if (!input) return;
  var rules = document.querySelectorAll('section.check details[data-search-text]');
  var debounceTimer = null;
  function apply(query) {
    var q = query.trim().toLowerCase();
    if (!q) {
      document.body.removeAttribute('data-filter-search');
      document.body.removeAttribute('data-filter-search-q');
      rules.forEach(function(r){ r.removeAttribute('data-match'); });
    } else {
      document.body.setAttribute('data-filter-search', '');
      document.body.setAttribute('data-filter-search-q', q);
      rules.forEach(function(r){
        var hay = r.getAttribute('data-search-text') || '';
        if (hay.indexOf(q) >= 0) r.setAttribute('data-match', '');
        else r.removeAttribute('data-match');
      });
    }
    if (window.__phase2_syncHash) window.__phase2_syncHash();
  }
  input.addEventListener('input', function(){
    clearTimeout(debounceTimer);
    var v = input.value;
    debounceTimer = setTimeout(function(){ apply(v); }, 150);
  });
  // Restore from hash on load.
  var hash = location.hash.replace(/^#/, '');
  if (hash) {
    var params = {};
    hash.split('&').forEach(function(p){
      var eq = p.indexOf('=');
      if (eq < 0) params[p] = true;
      else params[p.slice(0, eq)] = decodeURIComponent(p.slice(eq + 1));
    });
    if (params.search) {
      input.value = params.search;
      apply(params.search);
    }
  }
})();
```
